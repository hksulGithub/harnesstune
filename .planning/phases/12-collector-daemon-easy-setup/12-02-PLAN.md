<threat_model>
## Threat Model

**Assets at risk:**
- Relay token (in config file, env var, and launchd plist EnvironmentVariables)
- PID file at `~/.harnesstune/collector.pid` (process control)
- Collector status file (machine info disclosure)

**Threats:**
- PID file race condition: two `start` commands running simultaneously could both pass the stale-PID check
- Token in launchd plist readable by other users if plist has wrong permissions
- Unauthorized `stop` command kills collector via PID file
- Orphaned daemon if PID file deleted while process runs

**Mitigations:**
- PID duplicate detection via `process.kill(pid, 0)` before writing PID file (best-effort; not atomic lock, but matches established agent CLI pattern)
- Shutdown guard (`if (shuttingDown) return`) prevents double-execution of shutdown sequence
- Graceful signal handlers (`SIGTERM`, `SIGINT`, `SIGHUP`) send disconnected heartbeat + remove PID file
- `install` command sets plist file to user-only readable
- Token precedence: env var > config file — plist uses env var injection so token not duplicated in two places
</threat_model>

## Overview

Implement the CLI entry point with 5 subcommands: `setup` (guided onboarding), `start` (foreground daemon with heartbeat + plugin poll loop), `stop` (PID-based SIGTERM), `status` (PID liveness + status file read), and `install` (launchd plist generator). Also implement the daemon heartbeat and scheduler modules.

## must_haves

- `harnesstune-collector setup` prompts for relay URL, registers channel, detects platforms, writes `collector.json` with `chmod 600`
- `harnesstune-collector start` writes PID file, sends heartbeat every 5 minutes, runs plugin poll loop every 60s, writes `collector-status.json`
- `harnesstune-collector stop` reads PID file and sends SIGTERM
- `harnesstune-collector status` reports running/not-running and last heartbeat/poll timestamps
- `harnesstune-collector install` generates `~/Library/LaunchAgents/com.harnesstune.collector.plist`
- Graceful shutdown on SIGTERM/SIGINT/SIGHUP sends disconnected heartbeat and removes PID file

```typescript
import { randomUUID } from 'node:crypto';
import type { CollectorConfig, CollectorStatus } from '../config.js';
import { resolveToken } from '../config.js';
import type { RetryQueue } from '../queue.js';

export interface HeartbeatPluginSummary {
  [pluginId: string]: { enabled: boolean; agentCount: number };
}

export async function sendHeartbeat(
  config: CollectorConfig,
  queue: RetryQueue,
  status: 'connected' | 'disconnected',
  plugins: HeartbeatPluginSummary,
): Promise<void> {
  const token = resolveToken(config);
  const envelope = {
    type: 'heartbeat' as const,
    body: {
      status,
      uptimeSeconds: Math.floor(process.uptime()),
      plugins,
    },
    generatedAt: new Date().toISOString(),
    reportId: randomUUID(),
  };

  try {
    const res = await fetch(`${config.relayUrl}/channels/${config.channelId}/reports`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${token}`,
      },
      body: JSON.stringify(envelope),
    });
    if (!res.ok) {
      console.error(`Heartbeat upload failed: ${res.status}`);
      queue.enqueue(config.channelId, envelope);
    }
  } catch (err) {
    console.error('Heartbeat error:', err);
    queue.enqueue(config.channelId, envelope);
  }
}
```

```typescript
import { randomUUID } from 'node:crypto';
import type { CollectorConfig } from '../config.js';
import { resolveToken } from '../config.js';
import type { RetryQueue } from '../queue.js';
import type { PlatformPlugin } from '../plugins/interface.js';
import type { RunReport } from '@harnesstune/shared';

export interface PluginCursors {
  [pluginId: string]: Date;
}

export interface CycleResult {
  lastPoll: string;
  plugins: Record<string, { enabled: boolean; agentCount: number }>;
}

/**
 * Run one collection cycle: for each enabled plugin, call collectRuns(since),
 * upload results to relay, advance cursors.
 */
export async function runCycle(
  plugins: PlatformPlugin[],
  config: CollectorConfig,
  queue: RetryQueue,
  cursors: PluginCursors,
): Promise<CycleResult> {
  const token = resolveToken(config);
  const pluginSummary: Record<string, { enabled: boolean; agentCount: number }> = {};
  const enabledIds = new Set(config.platforms.filter(p => p.enabled).map(p => p.id));

  for (const plugin of plugins) {
    const enabled = enabledIds.has(plugin.id);
    if (!enabled) {
      pluginSummary[plugin.id] = { enabled: false, agentCount: 0 };
      continue;
    }

    try {
      // Discover agents for count
      const agents = await plugin.discover();
      pluginSummary[plugin.id] = { enabled: true, agentCount: agents.length };

      // Register discovered agents with relay
      for (const agent of agents) {
        try {
          await fetch(`${config.relayUrl}/channels/${config.channelId}/agents`, {
            method: 'POST',
            headers: {
              'Content-Type': 'application/json',
              'Authorization': `Bearer ${token}`,
            },
            body: JSON.stringify({
              agentId: agent.agentId,
              name: agent.name,
              platform: agent.platform,
              schedule: agent.schedule,
            }),
          });
        } catch (err) {
          console.error(`Failed to register agent ${agent.agentId}:`, err);
        }
      }

      // Collect runs since last cursor
      const since = cursors[plugin.id] ?? new Date(Date.now() - 7 * 24 * 60 * 60 * 1000);
      const runs = await plugin.collectRuns(since);

      // Upload each run to relay
      for (const run of runs) {
        try {
          const res = await fetch(`${config.relayUrl}/channels/${config.channelId}/runs`, {
            method: 'POST',
            headers: {
              'Content-Type': 'application/json',
              'Authorization': `Bearer ${token}`,
            },
            body: JSON.stringify(run),
          });
          if (!res.ok) {
            queue.enqueue(config.channelId, { type: 'run', run });
          }
        } catch {
          queue.enqueue(config.channelId, { type: 'run', run });
        }
      }

      // Advance cursor
      if (runs.length > 0) {
        const latest = runs.reduce((max, r) => {
          const t = new Date(r.finishedAt);
          return t > max ? t : max;
        }, since);
        cursors[plugin.id] = latest;
      }
    } catch (err) {
      console.error(`Plugin ${plugin.id} error:`, err);
      pluginSummary[plugin.id] = { enabled: true, agentCount: 0 };
    }
  }

  // Attempt queue replay on successful cycle (relay is reachable)
  const replayed = await queue.replay(config.relayUrl, token);
  if (replayed > 0) console.log(`Replayed ${replayed} queued report(s)`);

  return {
    lastPoll: new Date().toISOString(),
    plugins: pluginSummary,
  };
}
```

```typescript
import { createInterface } from 'node:readline/promises';
import { stdin as input, stdout as output } from 'node:process';
import { writeConfig } from '../config.js';
import type { CollectorConfig, PlatformEntry } from '../config.js';
import { ALL_PLUGINS } from '../plugins/loader.js';

function parseFlags(args: string[]): Record<string, string> {
  const flags: Record<string, string> = {};
  for (let i = 0; i < args.length; i++) {
    if (args[i].startsWith('--') && args[i + 1]) {
      flags[args[i].slice(2)] = args[++i];
    }
  }
  return flags;
}

export async function setup(args: string[]): Promise<void> {
  const flags = parseFlags(args);
  let relayUrl = flags['relay-url'];
  let machineName = flags['name'];

  const rl = createInterface({ input, output });
  try {
    relayUrl ??= await rl.question('Relay URL (e.g. https://harnesstune-relay.vercel.app/api): ');
    machineName ??= await rl.question('Machine name (e.g. mac-mini-office): ');

    relayUrl = relayUrl.replace(/\/$/, '');

    // Register channel with relay
    console.log('\nRegistering with relay...');
    const res = await fetch(`${relayUrl}/channels`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name: machineName }),
    });

    if (!res.ok) {
      const err = await res.json().catch(() => ({})) as { error?: string };
      console.error('Registration failed:', err.error ?? res.statusText);
      process.exit(1);
    }

    const data = await res.json() as { channelId: string; token: string; message: string };
    console.log(`Registered as channel ${data.channelId}`);
    console.log(`Token: ${data.token}`);
    console.log('(Save this token — it will not be shown again)\n');

    // Detect platforms
    console.log('Detecting installed platforms...');
    const platforms: PlatformEntry[] = [];
    for (const plugin of ALL_PLUGINS) {
      const detected = await plugin.detect();
      console.log(`  ${plugin.displayName}: ${detected ? 'DETECTED' : 'not found'}`);

      let enabled = detected;
      if (detected) {
        const answer = await rl.question(`  Enable ${plugin.displayName}? (Y/n): `);
        enabled = answer.trim().toLowerCase() !== 'n';
      }

      let pluginConfig: Record<string, unknown> = {};
      if (enabled) {
        pluginConfig = await plugin.setup();
      }

      platforms.push({
        id: plugin.id,
        enabled,
        config: pluginConfig,
      });
    }

    // Write config
    const config: CollectorConfig = {
      relayUrl,
      channelId: data.channelId,
      token: data.token,
      pollInterval: 60000,
      heartbeatInterval: 300000,
      platforms,
    };
    writeConfig(config);
    console.log('\nConfig written to ~/.harnesstune/collector.json (chmod 600)');
    console.log('Run: harnesstune-collector start');
  } finally {
    rl.close();
  }
}
```

```typescript
import { readConfig, writePid, removePid, readPid, writeStatus, resolveToken } from '../config.js';
import type { CollectorConfig, CollectorStatus } from '../config.js';
import { RetryQueue } from '../queue.js';
import { ALL_PLUGINS } from '../plugins/loader.js';
import { sendHeartbeat } from '../daemon/heartbeat.js';
import { runCycle } from '../daemon/scheduler.js';
import type { PluginCursors } from '../daemon/scheduler.js';
import type { PlatformPlugin } from '../plugins/interface.js';

const BACKOFF_INITIAL = 1000;
const BACKOFF_MAX = 5 * 60 * 1000;
const JITTER_MAX_MS = 5000;

export async function start(_args: string[], opts: { dryRun: boolean }): Promise<void> {
  const config = readConfig();
  const token = resolveToken(config);

  if (opts.dryRun) {
    console.log('Dry run — validating config...');
    console.log(`  Relay URL: ${config.relayUrl}`);
    console.log(`  Channel ID: ${config.channelId}`);
    console.log(`  Token: ${token.slice(0, 8)}...`);
    console.log(`  Poll interval: ${config.pollInterval}ms`);
    console.log(`  Heartbeat interval: ${config.heartbeatInterval}ms`);
    console.log(`  Platforms: ${config.platforms.filter(p => p.enabled).map(p => p.id).join(', ') || '(none)'}`);

    // Check relay reachability
    try {
      const res = await fetch(`${config.relayUrl.replace(/\/api$/, '')}/api/health`);
      console.log(`  Relay health: ${res.ok ? 'OK' : res.status}`);
    } catch (err) {
      console.error(`  Relay unreachable: ${err}`);
    }
    return;
  }

  // PID duplicate detection
  const existingPid = readPid();
  if (existingPid !== null) {
    try {
      process.kill(existingPid, 0);
      console.error(`Error: collector already running (PID ${existingPid}). Use 'harnesstune-collector stop' first.`);
      process.exit(1);
    } catch {
      console.warn(`Warning: stale PID file found for PID ${existingPid}, ignoring`);
      removePid();
    }
  }
  writePid(process.pid);

  const queue = new RetryQueue();
  const cursors: PluginCursors = {};

  // Filter to enabled plugins
  const enabledIds = new Set(config.platforms.filter(p => p.enabled).map(p => p.id));
  const enabledPlugins = ALL_PLUGINS.filter(p => enabledIds.has(p.id));

  // Build initial plugin summary
  function buildPluginSummary(): Record<string, { enabled: boolean; agentCount: number }> {
    const summary: Record<string, { enabled: boolean; agentCount: number }> = {};
    for (const p of ALL_PLUGINS) {
      summary[p.id] = { enabled: enabledIds.has(p.id), agentCount: 0 };
    }
    return summary;
  }

  // Graceful shutdown
  let shuttingDown = false;
  async function shutdown(): Promise<void> {
    if (shuttingDown) return;
    shuttingDown = true;
    console.log('Shutting down — sending disconnected heartbeat...');
    await sendHeartbeat(config, queue, 'disconnected', buildPluginSummary());
    removePid();
    process.exit(0);
  }
  process.on('SIGTERM', () => { void shutdown(); });
  process.on('SIGINT',  () => { void shutdown(); });
  process.on('SIGHUP',  () => { void shutdown(); });

  // Send initial connected heartbeat
  console.log(`Collector started (PID ${process.pid})`);
  console.log(`  Relay: ${config.relayUrl}`);
  console.log(`  Channel: ${config.channelId}`);
  console.log(`  Enabled plugins: ${enabledPlugins.map(p => p.displayName).join(', ') || '(none)'}`);
  console.log(`  Poll interval: ${config.pollInterval}ms`);
  console.log(`  Heartbeat interval: ${config.heartbeatInterval}ms`);

  await sendHeartbeat(config, queue, 'connected', buildPluginSummary());

  // Heartbeat timer (every heartbeatInterval, default 5 min)
  const heartbeatTimer = setInterval(() => {
    if (!shuttingDown) void sendHeartbeat(config, queue, 'connected', lastPluginSummary);
  }, config.heartbeatInterval);
  heartbeatTimer.unref();

  // Poll loop with exponential backoff + jitter
  let currentBackoff = BACKOFF_INITIAL;
  let lastPluginSummary = buildPluginSummary();

  async function pollLoop(): Promise<void> {
    if (shuttingDown) return;
    try {
      const result = await runCycle(enabledPlugins, config, queue, cursors);
      lastPluginSummary = result.plugins;

      // Write status file
      const statusData: CollectorStatus = {
        pid: process.pid,
        startedAt: startedAt,
        lastHeartbeat: new Date().toISOString(),
        lastPoll: result.lastPoll,
        plugins: result.plugins,
      };
      writeStatus(statusData);

      currentBackoff = BACKOFF_INITIAL;
    } catch (err) {
      console.error('Poll error:', err);
      currentBackoff = Math.min(currentBackoff * 2, BACKOFF_MAX);
    }

    if (!shuttingDown) {
      const baseDelay = currentBackoff === BACKOFF_INITIAL ? config.pollInterval : currentBackoff;
      const jitter = Math.floor(Math.random() * JITTER_MAX_MS);
      setTimeout(() => { void pollLoop(); }, baseDelay + jitter);
    }
  }

  const startedAt = new Date().toISOString();

  // Write initial status
  writeStatus({
    pid: process.pid,
    startedAt,
    lastHeartbeat: startedAt,
    lastPoll: null,
    plugins: buildPluginSummary(),
  });

  // Start first poll cycle
  void pollLoop();

  // Keep process alive
  await new Promise<void>((resolve) => {
    const keepAlive = setInterval(() => {
      if (shuttingDown) { clearInterval(keepAlive); resolve(); }
    }, 1000);
    keepAlive.unref();
  });
}
```

```typescript
import { readPid, removePid } from '../config.js';

export async function stop(_args: string[]): Promise<void> {
  const pid = readPid();
  if (!pid) {
    console.error('No running collector found (~/.harnesstune/collector.pid not present)');
    process.exit(1);
  }
  try {
    process.kill(pid, 'SIGTERM');
    console.log(`Sent SIGTERM to collector (PID ${pid})`);
  } catch (err: unknown) {
    if (err instanceof Error && 'code' in err && (err as NodeJS.ErrnoException).code === 'ESRCH') {
      console.error(`Process ${pid} not found -- removing stale PID file`);
      removePid();
    } else {
      throw err;
    }
  }
}
```

```typescript
import { readPid, readStatus } from '../config.js';

export async function status(_args: string[]): Promise<void> {
  const pid = readPid();
  let running = false;

  if (pid !== null) {
    try {
      process.kill(pid, 0);
      running = true;
    } catch {
      // ESRCH — process not found
    }
  }

  if (running) {
    console.log(`Collector is RUNNING (PID ${pid})`);
  } else if (pid !== null) {
    console.log(`Collector is NOT RUNNING (stale PID file for ${pid})`);
  } else {
    console.log('Collector is NOT RUNNING (no PID file)');
  }

  const statusData = readStatus();
  if (statusData) {
    const uptimeMs = Date.now() - new Date(statusData.startedAt).getTime();
    const uptimeHours = Math.floor(uptimeMs / 3600000);
    const uptimeMinutes = Math.floor((uptimeMs % 3600000) / 60000);

    console.log(`\nStarted at: ${statusData.startedAt}`);
    if (running) console.log(`Uptime: ${uptimeHours}h ${uptimeMinutes}m`);
    console.log(`Last heartbeat: ${statusData.lastHeartbeat ?? 'never'}`);
    console.log(`Last poll: ${statusData.lastPoll ?? 'never'}`);

    console.log('\nPlugins:');
    for (const [id, info] of Object.entries(statusData.plugins)) {
      const statusStr = info.enabled ? `enabled (${info.agentCount} agents)` : 'disabled';
      console.log(`  ${id}: ${statusStr}`);
    }
  } else {
    console.log('\nNo status data available (daemon may not have completed a cycle yet)');
  }
}
```

```typescript
const LAUNCH_AGENTS_DIR = join(homedir(), 'Library', 'LaunchAgents');
const PLIST_PATH = join(LAUNCH_AGENTS_DIR, `${PLIST_LABEL}.plist`);

export async function install(_args: string[]): Promise<void> {
```
