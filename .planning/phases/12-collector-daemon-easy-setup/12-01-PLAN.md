<threat_model>
## Threat Model

**Assets at risk:**
- Relay token in `collector.json` (API credential for relay)
- Config file at `~/.harnesstune/collector.json` (relay URL, channel ID)

**Threats:**
- Token leakage via world-readable config file
- Token leakage via console.log of config object

**Mitigations:**
- `writeConfig()` calls `chmodSync(CONFIG_FILE, 0o600)` — file only readable by owner
- `HARNESSTUNE_TOKEN` env var takes precedence over config file (D-05)
- Token field in `CollectorConfig` type is `string` — no accidental serialization beyond config write
</threat_model>

## Overview

Scaffold the `packages/harnesstune-collector` package, wire it into the monorepo, define the config module, retry queue, plugin interface, plugin loader, and all 4 stub plugins. This wave delivers the foundational types and infrastructure that the daemon commands (Wave 2) will consume.

- `readConfig()` / `writeConfig()` use `~/.harnesstune/collector.json` with `chmod 600`
- Token resolution checks `HARNESSTUNE_TOKEN` env var first, then config file
- Root `tsconfig.json` includes collector in `references`

import { existsSync, readFileSync, writeFileSync, mkdirSync, rmSync, chmodSync } from 'node:fs';
import { join } from 'node:path';
import { homedir } from 'node:os';

export const CONFIG_DIR = join(homedir(), '.harnesstune');
const CONFIG_FILE = join(CONFIG_DIR, 'collector.json');
export const PID_FILE = join(CONFIG_DIR, 'collector.pid');
const STATUS_FILE = join(CONFIG_DIR, 'collector-status.json');
const QUEUE_DIR = join(CONFIG_DIR, 'queue');

export interface PlatformEntry {
  id: string;
  enabled: boolean;
  config: Record<string, unknown>;
}

export interface CollectorConfig {
  relayUrl: string;
  channelId: string;
  token: string;
  pollInterval: number;       // ms — default 60000
  heartbeatInterval: number;  // ms — default 300000
  platforms: PlatformEntry[];
}

export interface CollectorStatus {
  pid: number;
  startedAt: string;
  lastHeartbeat: string | null;
  lastPoll: string | null;
  plugins: Record<string, { enabled: boolean; agentCount: number }>;
}

/** Read token with env var precedence (HARNESSTUNE_TOKEN > config file) */
export function resolveToken(config: CollectorConfig): string {
  return process.env['HARNESSTUNE_TOKEN'] ?? config.token;
}

export function readConfig(): CollectorConfig {
  if (!existsSync(CONFIG_FILE)) {
    throw new Error('No config found. Run: harnesstune-collector setup');
  }
  return JSON.parse(readFileSync(CONFIG_FILE, 'utf-8')) as CollectorConfig;
}

export function writeConfig(config: CollectorConfig): void {
  mkdirSync(CONFIG_DIR, { recursive: true });
  writeFileSync(CONFIG_FILE, JSON.stringify(config, null, 2), 'utf-8');
  chmodSync(CONFIG_FILE, 0o600);
}

export function writePid(pid: number): void {
  mkdirSync(CONFIG_DIR, { recursive: true });
  writeFileSync(PID_FILE, String(pid), 'utf-8');
}

export function readPid(): number | null {
  if (!existsSync(PID_FILE)) return null;
  const raw = readFileSync(PID_FILE, 'utf-8').trim();
  return raw ? parseInt(raw, 10) : null;
}

export function removePid(): void {
  if (existsSync(PID_FILE)) rmSync(PID_FILE);
}

export function writeStatus(status: CollectorStatus): void {
  mkdirSync(CONFIG_DIR, { recursive: true });
  writeFileSync(STATUS_FILE, JSON.stringify(status, null, 2), 'utf-8');
}

export function readStatus(): CollectorStatus | null {
  if (!existsSync(STATUS_FILE)) return null;
  return JSON.parse(readFileSync(STATUS_FILE, 'utf-8')) as CollectorStatus;
}

export function getQueueDir(): string {
  mkdirSync(QUEUE_DIR, { recursive: true });
  return QUEUE_DIR;
}

2. Create `packages/harnesstune-collector/src/queue.ts` — adapted from agent `queue.ts`:

```typescript
import { existsSync, mkdirSync, readFileSync, readdirSync, rmSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import { getQueueDir } from './config.js';

const MAX_QUEUE_SIZE = 48;
const REPLAY_MIN_INTERVAL_MS = 5000;

interface QueueEntry {
  path: string;
  channelId: string;
  body: unknown;
  timestamp: number;
}

export class RetryQueue {
  private readonly dir: string;

  constructor() {
    this.dir = getQueueDir();
  }

  enqueue(channelId: string, body: unknown): void {
    mkdirSync(this.dir, { recursive: true });
    const entries = this.list();
    if (entries.length >= MAX_QUEUE_SIZE) {
      const oldest = entries[0];
      if (oldest) rmSync(oldest.path);
    }
    const filename = `${Date.now()}-${Math.random().toString(36).slice(2, 8)}.json`;
    const filepath = join(this.dir, filename);
    writeFileSync(filepath, JSON.stringify({ channelId, body, timestamp: Date.now() }), 'utf-8');
  }

  list(): QueueEntry[] {
    if (!existsSync(this.dir)) return [];
    return readdirSync(this.dir)
      .filter(f => f.endsWith('.json'))
      .sort()
      .map(f => {
        const filepath = join(this.dir, f);
        const data = JSON.parse(readFileSync(filepath, 'utf-8')) as { channelId: string; body: unknown; timestamp: number };
        return { path: filepath, ...data };
      });
  }

  async replay(relayUrl: string, token: string): Promise<number> {
    const entries = this.list();
    if (entries.length === 0) return 0;
    let replayed = 0;
    for (const entry of entries) {
      try {
        const res = await fetch(`${relayUrl}/channels/${entry.channelId}/reports`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
          body: JSON.stringify(entry.body),
        });
        if (res.ok) {
          rmSync(entry.path);
          replayed++;
          if (replayed < entries.length) {
            await new Promise(resolve => setTimeout(resolve, REPLAY_MIN_INTERVAL_MS));
          }
        } else {
          break;
        }
      } catch {
        break;
      }
    }
    return replayed;
  }
}
```

3. Create `packages/harnesstune-collector/src/plugins/interface.ts`:

```typescript
import type { RunReport } from '@harnesstune/shared';

/** Per-platform config stored in collector.json platforms[].config */
export type PlatformConfig = Record<string, unknown>;

/** Agent identity as seen by the collector — matches relay POST /channels/:id/agents body shape */
export interface CollectorAgentIdentity {
  agentId: string;
  name: string | null;
  platform: string;
  schedule: string | null;
}

/** Contract for platform-specific data collection */
export interface PlatformPlugin {
  readonly id: string;
  readonly displayName: string;

  /** Detect if this platform is installed on this machine */
  detect(): Promise<boolean>;

  /** Interactive first-time config (readline prompts) */
  setup(): Promise<PlatformConfig>;

  /** Find all agents on this platform */
  discover(): Promise<CollectorAgentIdentity[]>;

  /** Collect completed runs since timestamp */
  collectRuns(since: Date): Promise<RunReport[]>;
}
```
